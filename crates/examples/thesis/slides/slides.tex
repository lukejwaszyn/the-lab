\begin{frame}{The Problem}
  \label{frame:problem}
  \begin{itemize}
    \item \TeX{} is a batch program: every run starts from the first byte.
    \item Authors edit one sentence at a time.
    \item Rebuild latency grows with document length, not edit size.
  \end{itemize}
\end{frame}

\begin{frame}{The Idea}
  \label{frame:idea}
  \begin{enumerate}
    \item Keep the engine alive between runs.
    \item Save a snapshot of its state every $k$ pages.
    \item After an edit, resume from the last snapshot before the change.
  \end{enumerate}
\end{frame}
